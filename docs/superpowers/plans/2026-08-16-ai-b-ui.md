```ts
/** 对应 Rust `AiStatus` */
export interface AiStatus {
  enabled: boolean
  configured: boolean
  /** 配置不完整时缺的那一项的中文名，完整时为 null */
  missingField: string | null
  pendingNotes: number
  indexedChunks: number
  memoryBytes: number
  /** 装载索引时跳过的维度不符行数，非 0 说明换过模型但没重建 */
  dimMismatches: number
  lastError: string | null
}

/** 对应 Rust `ConnectionReport` */
export interface ConnectionReport {
  embedOk: boolean
  embedDim: number | null
  chatOk: boolean
  error: string | null
}

/** 对应 Rust `Citation`。index 与回答里的 [n] 标注一一对应 */
export interface Citation {
  index: number
  noteId: number
  uuid: string
  title: string
  heading: string
  excerpt: string
}

/** 对应 Rust `ChatMessage` */
export interface ChatMessage {
  id: number
  role: 'user' | 'assistant'
  content: string
  citations: Citation[]
  createdAt: number
}

/** 对应 Rust `Conversation` */
export interface Conversation {
  id: number
  title: string
  createdAt: number
  updatedAt: number
}

/** 对应 Rust `SemanticHit` */
export interface SemanticHit {
  noteId: number
  uuid: string
  title: string
  excerpt: string
  score: number
}

/**
 * 对应 Rust `AskEvent`。serde 的默认外部标签序列化成
 * `{"Delta":{"text":"…"}}`，无字段的变体序列化成裸字符串 `"Cancelled"`。
 *
 * **这条路径不经过 `toCamel`**：Channel 的载荷由 Tauri 直接投递，
 * 所以下面的字段名必须写成 Rust 那边的 snake_case（`message_id`）。
 * 写成 camelCase 的话运行时拿到的是 undefined，界面会安静地不显示任何东西。
 */
export type AskEvent =
  | { Retrieved: { citations: Citation[] } }
  | { Delta: { text: string } }
  | { Done: { message_id: number } }
  | { Failed: { message: string } }
  | 'Cancelled'
```

再加命令封装（全部经 `toCamel`，除 `aiAsk` 外）：

```ts
export async function aiStatus(): Promise<AiStatus> {
  return toCamel(await invoke('ai_status'))
}

export async function aiEnable(enabled: boolean): Promise<{ pendingNotes: number }> {
  return toCamel(await invoke('ai_enable', { enabled }))
}

export async function aiTestConnection(): Promise<ConnectionReport> {
  return toCamel(await invoke('ai_test_connection'))
}

export async function aiReindexAll(): Promise<number> {
  return invoke('ai_reindex_all')
}

export async function aiRetryFailed(): Promise<number> {
  return invoke('ai_retry_failed')
}

/**
 * 发起提问。命令立刻返回，回答通过 channel 流式推来。
 * 调用方负责在组件卸载时调 `aiCancel`。
 */
export async function aiAsk(
  conversationId: number,
  question: string,
  onEvent: (event: AskEvent) => void,
): Promise<void> {
  const channel = new Channel<AskEvent>()
  channel.onmessage = onEvent
  await invoke('ai_ask', { conversationId, question, onEvent: channel })
}

export async function aiCancel(): Promise<void> {
  return invoke('ai_cancel')
}

export async function aiSemanticSearch(query: string, limit: number): Promise<SemanticHit[]> {
  return toCamel(await invoke('ai_semantic_search', { query, limit }))
}

export async function aiListConversations(limit: number, offset: number): Promise<Conversation[]> {
  return toCamel(await invoke('ai_list_conversations', { limit, offset }))
}

export async function aiCreateConversation(): Promise<number> {
  return invoke('ai_create_conversation')
}

export async function aiGetMessages(conversationId: number): Promise<ChatMessage[]> {
  return toCamel(await invoke('ai_get_messages', { conversationId }))
}

export async function aiDeleteConversation(id: number): Promise<void> {
  return invoke('ai_delete_conversation', { id })
}

export async function aiRenameConversation(id: number, title: string): Promise<void> {
  return invoke('ai_rename_conversation', { id, title })
}
```

`Channel` 从 `@tauri-apps/api/core` 导入，与 `invoke` 同一处。

import { describe, expect, it } from 'vitest'
import { formatBytes, reduceAsk, initialAsk, splitCitedText } from './ai'
import type { AskEvent } from './ipc'

const citation = (index: number, noteId: number) => ({
  index, noteId, uuid: `u${noteId}`, title: `笔记${noteId}`, heading: '', excerpt: '片段',
})

function play(events: AskEvent[]) {
  return events.reduce(reduceAsk, initialAsk())
}

describe('reduceAsk', () => {
  it('把多个 Delta 拼成完整回答', () => {
    const state = play([
      { Retrieved: { citations: [citation(1, 7)] } },
      { Delta: { text: '知识' } },
      { Delta: { text: '图谱' } },
      { Done: { message_id: 42 } },
    ])
    expect(state.answer).toBe('知识图谱')
    expect(state.phase).toBe('done')
    expect(state.messageId).toBe(42)
  })

  // 引用要在模型开口之前就亮出来——这是「可核验」这条主张的实现方式。
  it('Retrieved 一到就有引用，此时还没有任何回答文字', () => {
    const state = play([{ Retrieved: { citations: [citation(1, 7)] } }])
    expect(state.citations).toHaveLength(1)
    expect(state.answer).toBe('')
    expect(state.phase).toBe('streaming')
  })

  it('Failed 记下错误并结束', () => {
    const state = play([{ Delta: { text: '半截' } }, { Failed: { message: '网络断了' } }])
    expect(state.phase).toBe('failed')
    expect(state.error).toBe('网络断了')
  })

  // 取消不是错误：不能弹错误横幅，但必须收起「正在思考」。
  it('Cancelled 结束但不产生错误', () => {
    const state = play([{ Delta: { text: '半截' } }, 'Cancelled'])
    expect(state.phase).toBe('cancelled')
    expect(state.error).toBeNull()
  })

  // Done 之后再来的事件必须被忽略，否则一次迟到的 Delta 会把
  // 已经落库的回答又改一遍，界面与数据库从此对不上。
  it('终止之后的事件被忽略', () => {
    const state = play([
      { Done: { message_id: 1 } },
      { Delta: { text: '迟到的' } },
      { Failed: { message: '迟到的错误' } },
    ])
    expect(state.answer).toBe('')
    expect(state.phase).toBe('done')
    expect(state.error).toBeNull()
  })
})

describe('splitCitedText', () => {
  it('把 [n] 标注切出来，编号能对上引用', () => {
    const parts = splitCitedText('这是结论[1]，另一条[2]。')
    expect(parts.filter((p) => p.kind === 'cite').map((p) => p.index)).toEqual([1, 2])
    expect(parts.map((p) => (p.kind === 'text' ? p.text : '')).join('')).toBe('这是结论，另一条。')
  })

  it('没有标注时原样返回一段文本', () => {
    expect(splitCitedText('纯文本')).toEqual([{ kind: 'text', text: '纯文本' }])
  })

  // 模型偶尔会写 [1][2] 连着，或者写出根本不存在的 [9]。
  // 前者要拆成两个，后者不能崩——渲染成普通文本即可。
  it('处理连续标注与越界编号', () => {
    const parts = splitCitedText('结论[1][2]，还有[9]')
    expect(parts.filter((p) => p.kind === 'cite').map((p) => p.index)).toEqual([1, 2, 9])
  })

  it('方括号里不是数字时当普通文本', () => {
    expect(splitCitedText('数组[i]取值')).toEqual([{ kind: 'text', text: '数组[i]取值' }])
  })
})

describe('formatBytes', () => {
  it('给出人能读的内存占用', () => {
    expect(formatBytes(0)).toBe('0 B')
    expect(formatBytes(1024)).toBe('1.0 KB')
    expect(formatBytes(120 * 1024 * 1024)).toBe('120.0 MB')
  })
})

- [ ] **Step 3: 实现 `ui/lib/ai.ts`**

要点：

- `AskState = { phase: 'idle' | 'streaming' | 'done' | 'failed' | 'cancelled', answer: string, citations: Citation[], messageId: number | null, error: string | null }`
- `initialAsk(): AskState` 返回 `phase: 'idle'`，其余为空。
- `reduceAsk(state, event)`：**终止态（done/failed/cancelled）下直接返回原 state**。这条不是防御性编程——网络包的到达顺序不保证，一次迟到的 Delta 会把已经落库的回答又改一遍，界面与数据库从此对不上。
- `splitCitedText(text)`：用 `/\[(\d+)\]/g` 切分，产出 `{ kind: 'text', text } | { kind: 'cite', index }`。方括号里不是纯数字的原样保留。
- `formatBytes(n)`：B / KB / MB / GB，保留一位小数（0 B 除外）。

- [ ] **Step 1: 给 mock 加 `Channel` 支持**

**保真度是这里的全部价值。** `@tauri-apps/api` 2.11 的 `Channel` 在被当作 invoke 参数序列化时，会走 `__TAURI_INTERNALS__.transformCallback` 拿到一个数字 id，并序列化成 `__CHANNEL__:<id>` 形式的字符串（具体形状请**打开 `node_modules/@tauri-apps/api/core.js` 核对**，不要凭记忆写）。mock 必须：

1. 让 `transformCallback` 真的登记回调并返回 id；
